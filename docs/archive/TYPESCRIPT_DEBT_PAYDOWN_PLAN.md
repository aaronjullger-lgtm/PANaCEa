## 🧬 The Prisma TS2322 Problem Explained

### What is Prisma's `Exact<>` Type?

Prisma v7 generates strict input types that use the `Exact<>` utility type:

```typescript
// Generated by Prisma
type Exact<InputType, Shape> = InputType extends Shape ? Shape : never;

// Example: UserAchievementCreateInput
type UserAchievementCreateInput = Exact<
  {
    id: string;
    userId: string;
    achievementId: string;
    unlockedAt?: DateTime;
    progress?: number;
    User: UserCreateNestedOneWithoutUserAchievementInput;
  },
  ActualShape
>;
```

**The Problem:**

```typescript
// ❌ FAILS - Missing required relation field 'User'
await prisma.userAchievement.create({
  data: {
    userId: '123', // Field is present
    achievementId: 'abc', // Field is present
    progress: 100, // Field is present
    // ❌ Missing 'User' relation field or 'id' field
  },
});

// ✅ PASSES - Option 1: Include relation
await prisma.userAchievement.create({
  data: {
    id: generateId(),
    userId: '123',
    achievementId: 'abc',
    progress: 100,
    User: {
      connect: { id: '123' },
    },
  },
});

// ✅ PASSES - Option 2: Use UncheckedCreateInput (bypasses relations)
await prisma.userAchievement.create({
  data: {
    id: generateId(),
    userId: '123', // Foreign key directly
    achievementId: 'abc',
    progress: 100,
  } as Prisma.UserAchievementUncheckedCreateInput,
});
```

### Generic Strategies for Fixing Prisma TS2322 Errors

#### Strategy 1: **Use Unchecked Input Types** (Recommended)

```typescript
import { Prisma } from '@prisma/client';

// ✅ Bypasses relation requirements
const data: Prisma.UserAchievementUncheckedCreateInput = {
  id: generateId(),
  userId: userId,
  achievementId: achievementId,
  progress: 100,
};

await prisma.userAchievement.create({ data });
```

**When to use:**

- You only have foreign key values (not relation objects)
- You want to avoid nested relation syntax
- 90% of our cases fall here

#### Strategy 2: **Use Relation Connect/Create**

```typescript
// ✅ When you need to validate relations exist
await prisma.userAchievement.create({
  data: {
    id: generateId(),
    achievementId: 'abc',
    progress: 100,
    User: {
      connect: { id: userId }, // Validates user exists
    },
  },
});
```

**When to use:**

- You need Prisma to validate the related record exists
- You're creating nested relations simultaneously
- 10% of our cases

#### Strategy 3: **Add Missing Required Fields**

```typescript
// Check schema for required fields without defaults
model UserAchievement {
  id            String   @id  // ⚠️ Required, no @default()
  userId        String        // ⚠️ Required
  achievementId String        // ⚠️ Required
  progress      Int      @default(100) // ✅ Has default
}

// Must provide 'id' explicitly
await prisma.userAchievement.create({
  data: {
    id: generateId(), // ⚠️ MUST include
    userId: "123",
    achievementId: "abc",
    // progress is optional due to @default(100)
  } as Prisma.UserAchievementUncheckedCreateInput
});
```

#### Strategy 4: **Create DTO Adapters for Complex Types**


**When to use:**

- Frontend/API boundary types differ from Prisma schema
- Multiple services use the same create pattern
- Need validation/transformation logic
- 20-30% of complex cases


### Strategic Decision: Update Code, Not Schema

**Why not update the Prisma schema?**

- Schema is production database structure
- Changing schema requires migrations that could break existing data
- Would require coordinating database + code changes simultaneously
- Risk of data loss or migration failures

**Why update the code instead?**

- Code is more flexible and easier to iterate on
- Can use TypeScript features (adapters, type assertions) to bridge gaps
- Doesn't require database changes
- Can be done incrementally without risk

**Answer: Update code to match Prisma's strict types**

```typescript
import { Prisma } from '@prisma/client';
import { randomBytes } from 'crypto';

/**
 * Generate unique IDs for Prisma models
 */
export function generateId(): string {
  return randomBytes(16).toString('hex');
}

/**
 * Type-safe wrapper for Unchecked create operations
 */
export type UncheckedCreate<T extends keyof Prisma.TypeMap['model']> =
  Prisma.TypeMap['model'][T]['operations']['create']['args']['data'];

/**
 * Helper to ensure required fields are present
 */
export function validateRequiredFields<T extends Record<string, any>>(
  data: T,
  requiredFields: (keyof T)[]
): T {
  for (const field of requiredFields) {
    if (data[field] === undefined || data[field] === null) {
      throw new Error(`Missing required field: ${String(field)}`);
    }
  }
  return data;
}

/**
 * Safe date conversion
 */
export function toDate(value: number | Date | string): Date {
  if (value instanceof Date) return value;
  if (typeof value === 'number') return new Date(value);
  return new Date(value);
}
```

## 🔧 Code Mod Strategy

### Automated Bulk Fix Script

**Location:** `scripts/fix-prisma-creates.ts`

```typescript
import * as ts from 'typescript';
import * as fs from 'fs';
import * as path from 'path';

interface PrismaCreateFix {
  filePath: string;
  line: number;
  model: string;
  currentDataObject: string;
  suggestedFix: string;
}

/**
 * Finds all prisma.*.create() calls and suggests fixes
 */
function findPrismaCreates(sourceFile: ts.SourceFile): PrismaCreateFix[] {
  const fixes: PrismaCreateFix[] = [];

  function visit(node: ts.Node) {
    // Look for: prisma.MODEL.create({ data: {...} })
    if (
      ts.isCallExpression(node) &&
      ts.isPropertyAccessExpression(node.expression) &&
      node.expression.name.text === 'create'
    ) {
      const prismaAccess = node.expression.expression;
      if (
        ts.isPropertyAccessExpression(prismaAccess) &&
        ts.isIdentifier(prismaAccess.expression) &&
        prismaAccess.expression.text === 'prisma'
      ) {
        const model = prismaAccess.name.text;
        const dataArg = node.arguments[0];

        if (ts.isObjectLiteralExpression(dataArg)) {
          const dataProperty = dataArg.properties.find(
            (p) => ts.isPropertyAssignment(p) && ts.isIdentifier(p.name) && p.name.text === 'data'
          );

          if (dataProperty && ts.isPropertyAssignment(dataProperty)) {
            const currentData = dataProperty.initializer.getText(sourceFile);
            const lineNumber = sourceFile.getLineAndCharacterOfPosition(node.getStart()).line + 1;

            fixes.push({
              filePath: sourceFile.fileName,
              line: lineNumber,
              model: model,
              currentDataObject: currentData,
              suggestedFix: generateAdapterFunction(model, currentData),
            });
          }
        }
      }
    }

    ts.forEachChild(node, visit);
  }

  visit(sourceFile);
  return fixes;
}

/**
 * Generates adapter function suggestion
 */
function generateAdapterFunction(model: string, dataObject: string): string {
  // Parse the object to extract fields
  const fields = extractFieldsFromObject(dataObject);
  const functionName = `create${model}Data`;

  const params = fields.map((f) => `${f.name}: ${f.type}`).join(', ');
  const returnType = `Prisma.${model}UncheckedCreateInput`;

  return `
function ${functionName}(${params}): ${returnType} {
  return {
    id: generateId(),
    ${fields.map((f) => `${f.name}: ${f.name},`).join('\n    ')}
  };
}

// Replace prisma.${model}.create call with:
await prisma.${model}.create({
  data: ${functionName}(${fields.map((f) => f.name).join(', ')})
});
`;
}

function extractFieldsFromObject(obj: string): Array<{ name: string; type: string }> {
  // Simple regex-based extraction (improve with actual AST parsing)
  const fieldRegex = /(\w+):\s*([^,\n]+)/g;
  const fields: Array<{ name: string; type: string }> = [];
  let match;

  while ((match = fieldRegex.exec(obj)) !== null) {
    fields.push({
      name: match[1],
      type: inferType(match[2]),
    });
  }

  return fields;
}

function inferType(value: string): string {
  if (value.match(/^['"`]/)) return 'string';
  if (value.match(/^\d+$/)) return 'number';
  if (value === 'true' || value === 'false') return 'boolean';
  if (value.match(/^new Date/)) return 'Date';
  if (value.match(/^\[/)) return 'any[]';
  return 'any';
}

/**
 * Scan all TypeScript files
 */
function scanProject(rootDir: string) {
  const allFixes: PrismaCreateFix[] = [];

  function scanDir(dir: string) {
    const entries = fs.readdirSync(dir, { withFileTypes: true });

    for (const entry of entries) {
      const fullPath = path.join(dir, entry.name);

      if (entry.isDirectory() && !entry.name.startsWith('.') && entry.name !== 'node_modules') {
        scanDir(fullPath);
      } else if (entry.isFile() && entry.name.endsWith('.ts')) {
        const content = fs.readFileSync(fullPath, 'utf-8');
        const sourceFile = ts.createSourceFile(fullPath, content, ts.ScriptTarget.Latest, true);

        const fixes = findPrismaCreates(sourceFile);
        allFixes.push(...fixes);
      }
    }
  }

  scanDir(rootDir);
  return allFixes;
}

// Run the scan
const projectRoot = process.cwd();
const fixes = scanProject(projectRoot);

console.log(`\n🔍 Found ${fixes.length} prisma.*.create() calls\n`);

// Group by model
const byModel = fixes.reduce(
  (acc, fix) => {
    if (!acc[fix.model]) acc[fix.model] = [];
    acc[fix.model].push(fix);
    return acc;
  },
  {} as Record<string, PrismaCreateFix[]>
);

// Output report
for (const [model, modelFixes] of Object.entries(byModel)) {
  console.log(`\n📦 ${model} (${modelFixes.length} calls)`);
  console.log('─'.repeat(50));

  for (const fix of modelFixes) {
    console.log(`\n📄 ${fix.filePath}:${fix.line}`);
    console.log(`\n${fix.suggestedFix}`);
  }
}

// Save to file
fs.writeFileSync('prisma-create-fixes.json', JSON.stringify(fixes, null, 2));

console.log(`\n✅ Report saved to prisma-create-fixes.json`);
```
